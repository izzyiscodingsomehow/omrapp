const functions = require("firebase-functions");
const nodemailer = require("nodemailer");

const transporter = nodemailer.createTransport({
  service: "gmail",
  auth: {
    user: "socialomr123@gmail.com",
    pass: "your-email-password",
  },
});

exports.sendEmail = functions.https.onRequest(async (req, res) => {
  const userdoc  = req.body;

  const emailBody = `
    User Name: ${userdoc.username}
    Loader Loader: ${userdoc.loadername}
    User Phone: ${userdoc.userphone}
    Loader Phone: ${userdoc.loaderphone}
    Created At: ${userdoc.createdAt}
    Cost: ${userdoc.cost}
  `;

  const mailOptions = {
    from: "socialomr123@gmail.com",
    to: "socialomr123@gmail.com",
    subject: userdoc.userid,
    text: emailBody,
  };

  try {
    await transporter.sendMail(mailOptions);
    res.status(200).send("Email sent successfully");
  } catch (error) {
    console.error(error);
    res.status(500).send("Internal Server Error");
  }
  
});
